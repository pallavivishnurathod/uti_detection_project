import os, json, warnings
warnings.filterwarnings("ignore")

import numpy as np
import pandas as pd

from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout, BatchNormalization
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import EarlyStopping
from sklearn.metrics import classification_report, confusion_matrix, roc_auc_score

import joblib
import gradio as gr
from google.colab import drive

from llama_cpp import Llama

drive.mount("/content/drive")

CSV_PATH      = "/content/drive/MyDrive/uti.csv"              # your dataset

df = pd.read_csv(CSV_PATH)
print("Loaded dataset shape:", df.shape)

drop_cols = [
    "ID", "PATID", "lang", "race", "ethnicity", "maritalStatus", "employStatus",
    "insurance_status", "disposition", "arrival", "dispo", "split", "abxUTI",
    "alt_diag", "O2_Amount_Last", "O2_Amount_First", "O2_Amount_Max",
    "O2_Amount_Min", "O2_Dependency_First",
    "O2_Dependency_Last", "alert"
]
df.drop(columns=[c for c in drop_cols if c in df.columns], inplace=True, errors="ignore")

# Drop columns with >75% 'not_reported' or null
columns_to_drop_not_reported = []
threshold = 0.75 * len(df)

for col in df.columns:
    not_reported_count = df[col].astype(str).str.strip().str.lower().eq("not_reported").sum()
    null_count = df[col].isnull().sum()
    if (not_reported_count + null_count) > threshold:
        columns_to_drop_not_reported.append(col)

if columns_to_drop_not_reported:
    df.drop(columns=columns_to_drop_not_reported, inplace=True, errors="ignore")
    print("Dropped sparse columns:", len(columns_to_drop_not_reported))

# Target mapping
df["UTI_diag"] = df["UTI_diag"].astype(str).str.strip().str.lower().map({"yes": 1, "no": 0})
df = df[df["UTI_diag"].notnull()]
print("After dropping rows without target:", df.shape)

# Separate features/target (keep raw df copy for manual row building)
target = df["UTI_diag"]
df_features = df.drop(columns=["UTI_diag"])

# Identify categorical & numeric
num_cols = df_features.select_dtypes(include=[np.number]).columns.tolist()
cat_cols = df_features.select_dtypes(include=["object", "category"]).columns.tolist()
print(f"Numeric cols: {len(num_cols)}, Categorical cols: {len(cat_cols)}")

# Impute
cat_imputer = SimpleImputer(strategy="most_frequent")
num_imputer = SimpleImputer(strategy="median")

if cat_cols:
    df_features[cat_cols] = cat_imputer.fit_transform(df_features[cat_cols])
if num_cols:
    df_features[num_cols] = num_imputer.fit_transform(df_features[num_cols])

# One-hot encode categoricals
df_encoded = pd.get_dummies(df_features, columns=cat_cols, drop_first=True, dtype=np.int64)

# Add target
df_processed = df_encoded.copy()
df_processed["UTI_diag"] = target.values

# Ensure numeric and fill NA
df_processed = df_processed.apply(pd.to_numeric, errors="coerce").fillna(0).astype(np.float32)

print("Processed data shape:", df_processed.shape)

X = df_processed.drop(columns=["UTI_diag"])
y = df_processed["UTI_diag"]

scaler = StandardScaler()
numeric_cols_after = X.select_dtypes(include=[np.number]).columns.tolist()
X[numeric_cols_after] = scaler.fit_transform(X[numeric_cols_after])

joblib.dump(scaler, SCALER_PATH)
print("Saved scaler to:", SCALER_PATH)

train_columns = X.columns.tolist()
with open(COLS_PATH, "w") as f:
    json.dump(train_columns, f)
print("Saved train columns to:", COLS_PATH)

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.2, stratify=y, random_state=42
)
print("Train/Test shapes:", X_train.shape, X_test.shape)

input_dim = X_train.shape[1]
mlp_model = Sequential([
    Dense(256, activation="relu", input_dim=input_dim),
    BatchNormalization(),
    Dropout(0.3),
    Dense(128, activation="relu"),
    BatchNormalization(),
    Dropout(0.3),
    Dense(64, activation="relu"),
    Dropout(0.2),
    Dense(1, activation="sigmoid")
])

mlp_model.compile(
    optimizer=Adam(learning_rate=0.001),
    loss="binary_crossentropy",
    metrics=["accuracy"]
)

early_stop = EarlyStopping(monitor="val_loss", patience=10, restore_best_weights=True)

history = mlp_model.fit(
    X_train, y_train,
    validation_data=(X_test, y_test),
    epochs=50,
    batch_size=64,
    callbacks=[early_stop],
    verbose=1
)

y_pred_prob = mlp_model.predict(X_test).ravel()
y_pred = (y_pred_prob > 0.5).astype(int)

print("\nClassification Report:\n", classification_report(y_test, y_pred))
print("\nConfusion Matrix:\n", confusion_matrix(y_test, y_pred))
try:
    print("\nROC-AUC:", roc_auc_score(y_test, y_pred_prob))

    pass

def build_full_input(manual_dict):
    """
    Build a single encoded & scaled row for the MLP using:
    - raw df column structure
    - same encoding & scaling as training
    """
    base = {}
    for col in df.columns:
        if col == "UTI_diag":
            continue
        try:
            base[col] = df[col].mode().iloc[0]
        except Exception:
            base[col] = "not_reported"

    for k, v in manual_dict.items():
        base[k] = v

    row = pd.DataFrame([base], columns=[c for c in df.columns if c != "UTI_diag"])

    # Fill missing like training
    for c in row.select_dtypes(include=["object","category"]).columns:
        if c in df.columns:
            row[c] = row[c].fillna(df[c].mode().iloc[0])
        else:
            row[c] = row[c].fillna("not_reported")

    for c in row.select_dtypes(include=[np.number]).columns:
        if c in df.columns:
            # Ensure column is numeric before calculating median
            row[c] = row[c].fillna(pd.to_numeric(df[c], errors='coerce').median())
        else:
            row[c] = row[c].fillna(0)

    # One-hot like training
    cat_like = row.select_dtypes(include=["object","category"]).columns.tolist()
    row_encoded = pd.get_dummies(row, columns=cat_like, drop_first=True, dtype=np.int64)

    for col in train_columns:
        if col not in row_encoded.columns:
            row_encoded[col] = 0

    row_encoded = row_encoded[train_columns]
    row_encoded[numeric_cols_after] = scaler.transform(row_encoded[numeric_cols_after])
    return row_encoded

    label = "UTI Detected" if prob > 0.5 else "No UTI Detected"
    return f"{label} (probability: {prob:.3f})"

SYSTEM_PROMPT = """
You are a friendly UTI medical assistant. You must ONLY talk about:

• Urinary Tract Infection (UTI)
• Symptoms, causes and common risk factors
• Prevention habits and hygiene
• Basic treatment guidance and self-care
• When to see a doctor or seek urgent help
• Tests commonly used to check for UTI
• General lifestyle guidance for recovery

STRICT RULES:
• Do NOT mention pregnancy, sexual activity, condoms, childbirth, fertility, or relationships.
• Do NOT discuss diseases other than UTI.
• Do NOT prescribe medicines, antibiotics, brand names, or dosages.
• Keep every answer safe, simple, and non-clinical.

CONVERSATION RULES:
• If the user asks about treatment or prevention, check whether AGE and GENDER are provided.
• If missing, ask for age and gender using one short follow-up question.
• Use the provided age and gender to give a personalized BUT general plan.
• If the user asks something unrelated to UTI, politely redirect them back to urinary infection topics.

STYLE:
• Use short sentences, easy English, and step-by-step guidance.
• NEVER pretend to be a doctor.
• End every answer with a short disclaimer such as:
  “Please consult a healthcare professional for medical diagnosis or treatment.”
"""

def ask_uti_bot(user_msg, gender, age):
    profile = f"Patient profile: gender={gender}, age={age}."
    full_prompt = f"<s>[INST]{SYSTEM_PROMPT}\n{profile}\nUser: {user_msg}[/INST]"

    response = llm(
        full_prompt,
        max_tokens=200,
        temperature=0.4,
        top_p=0.9
    )

    answer = response["choices"][0]["text"].strip()

    if "disclaimer" not in answer.lower():
        answer += "\n\nDisclaimer: This is general educational information. Please consult a doctor for proper care."

    return answer
